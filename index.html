// Forçar recalcular a planilha (resolve GOOGLEFINANCE congelado) // Configurar: Acionadores → Adicionar → forceRefresh → Cronometrado → A cada hora function forceRefresh() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var range = sheet.getDataRange(); range.getValues(); // Força leitura, dispara recálculo do GOOGLEFINANCE SpreadsheetApp.flush(); // Garante que o cache é atualizado } function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var action = e.parameter.action || 'add'; var ticker = (e.parameter.ticker || '').toUpperCase().trim(); if (action === 'add' && ticker) { // Verificar se já existe var data = sheet.getDataRange().getValues(); for (var i = 0; i < data.length; i++) { if (String(data[i][0]).toUpperCase().trim() === ticker) { return ContentService.createTextOutput(JSON.stringify({status:'exists'})) .setMimeType(ContentService.MimeType.JSON); } } var lastRow = sheet.getLastRow() + 1; sheet.getRange(lastRow, 1).setValue(ticker); setFormulas(sheet, lastRow, ticker); return ContentService.createTextOutput(JSON.stringify({status:'added', ticker: ticker})) .setMimeType(ContentService.MimeType.JSON); } if (action === 'sync') { var tickers = (e.parameter.tickers || '').split(',').filter(function(t) { return t.trim(); }); var existing = sheet.getDataRange().getValues().map(function(r) { return String(r[0]).toUpperCase().trim(); }); var added = 0; tickers.forEach(function(ticker) { ticker = ticker.toUpperCase().trim(); if (!ticker || existing.indexOf(ticker) >= 0) return; var lastRow = sheet.getLastRow() + 1; sheet.getRange(lastRow, 1).setValue(ticker); setFormulas(sheet, lastRow, ticker); existing.push(ticker); added++; }); return ContentService.createTextOutput(JSON.stringify({status:'synced', added: added})) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({status:'error', msg:'missing params'})) .setMimeType(ContentService.MimeType.JSON); } function setFormulas(sheet, row, ticker) { var cellRef = 'A' + row; if (ticker === 'USDBRL') { sheet.getRange(row, 2).setFormula('=GOOGLEFINANCE("CURRENCY:USDBRL")'); sheet.getRange(row, 3).setFormula('=(GOOGLEFINANCE("CURRENCY:USDBRL") - INDEX(GOOGLEFINANCE("CURRENCY:USDBRL","close",TODAY()-1,TODAY()-1),2,2))/INDEX(GOOGLEFINANCE("CURRENCY:USDBRL","close",TODAY()-1,TODAY()-1),2,2)'); } else if (ticker === 'BTCBRL') { sheet.getRange(row, 2).setFormula('=GOOGLEFINANCE("CURRENCY:BTCBRL")'); sheet.getRange(row, 3).setFormula('=(GOOGLEFINANCE("CURRENCY:BTCBRL") - INDEX(GOOGLEFINANCE("CURRENCY:BTCBRL","close",TODAY()-1,TODAY()-1),2,2))/INDEX(GOOGLEFINANCE("CURRENCY:BTCBRL","close",TODAY()-1,TODAY()-1),2,2)'); } else if (ticker === 'ETHBRL') { sheet.getRange(row, 2).setFormula('=GOOGLEFINANCE("CURRENCY:ETHBRL")'); sheet.getRange(row, 3).setFormula('=(GOOGLEFINANCE("CURRENCY:ETHBRL") - INDEX(GOOGLEFINANCE("CURRENCY:ETHBRL","close",TODAY()-1,TODAY()-1),2,2))/INDEX(GOOGLEFINANCE("CURRENCY:ETHBRL","close",TODAY()-1,TODAY()-1),2,2)'); } else { // Ações, FIIs, ETFs sheet.getRange(row, 2).setFormula('=GOOGLEFINANCE(CONCATENATE("bvmf:",' + cellRef + '),"price")'); sheet.getRange(row, 3).setFormula('=GOOGLEFINANCE(CONCATENATE("bvmf:",' + cellRef + '),"changepct")/100'); // Colunas opcionais: P/L e P/VP sheet.getRange(row, 4).setFormula('=GOOGLEFINANCE(CONCATENATE("bvmf:",' + cellRef + '),"pe")'); sheet.getRange(row, 5).setFormula('=GOOGLEFINANCE(CONCATENATE("bvmf:",' + cellRef + '),"pricetobook")'); } }
